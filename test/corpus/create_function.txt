   @pJob JobType
  ,@pSuffix SuffixType
)
RETURNS Flag
AS
BEGIN
   declare @Flag ListYesNoType

   set @Flag = case
      when exists(select 1 from ORDIND000 WITH (NOLOCK)
         where ORDIND000.ORDERID = dbo.ApsJobOrderId(@pJob, @pSuffix))
         then 1
         else 0
      end

   return @Flag
END;

CREATE FUNCTION dbo.CalcTotals (@CalcType GenericCodeType, @Type ProjCostTypeType)
RETURNS AmountType
AS
BEGIN
   DECLARE @ReturnValue AmountType, @TempAct AmountType

   SET @ReturnValue = 0.0

   IF @CalcType = N'OV_ACT' AND @Type = 'P'
   BEGIN
      SELECT @ReturnValue = SUM( pcst.ovh_a_cost )
      FROM projcost pcst
      WHERE pcst.type = @Type
   END

   -- Overhead Forecast Cost
   ELSE IF @CalcType = N'OV_FCT'
   BEGIN
      IF @Type = 'P'
         SELECT @ReturnValue = SUM( pcst.a_cost ) + SUM( pcst.ovh_a_cost )
         FROM projcost pcst
         WHERE pcst.type = @Type
      ELSE
         SELECT @ReturnValue = SUM( pcst.a_cost )
         FROM projcost pcst
   END

   IF @ReturnValue IS NULL
      SET @ReturnValue = 0.0

   RETURN ROUND(@ReturnValue, 2)
END
