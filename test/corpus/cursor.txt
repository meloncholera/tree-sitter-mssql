  SELECT item, qty FROM dbo.items WHERE site = @site
OPEN item_cursor
FETCH NEXT FROM item_cursor INTO @item, @qty
WHILE @@FETCH_STATUS = 0
BEGIN
  EXEC dbo.Process @item, @qty
  FETCH NEXT FROM item_cursor INTO @item, @qty
END
CLOSE item_cursor

CLOSE cur;
